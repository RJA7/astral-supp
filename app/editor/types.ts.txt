// Global / Generic Properties
// "path": The full path from the project root (e.g., /assets/images/hero.atlas).
// "text": The raw text content of the file. Important: This includes unsaved changes currently in the editor.
// "children": For directories, returns a list of child resource paths. For collections/GUIs, returns a list of internal node IDs.
//
// Collection & Game Object Properties
// "id": The string ID of the Game Object (e.g., "enemy").
// "position": The vec3 coordinates.
// "rotation": The quat or vec3 rotation.
// "scale": The vec3 scale.
// "components": A list of node IDs for components attached to a Game Object.
// "instance": For referenced GOs in a collection, the path to the .go file.
//
// Component Specific Properties
// "script": The path to the .script file for a script component.
// "image" / "default_animation": Common for Sprites.
// "material": The path to the .material file.
// "shapes": For Collision Objects, returns a list of collision shape nodes.
//
// UI & Atlas Properties
// "nodes": (GUI only) A list of all GUI nodes (box, text, etc.).
// "images": (Atlas only) A list of all individual image nodes in the atlas.
// "animations": (Atlas only) A list of animation nodes.
// "layers": (Tilemap/GUI) A list of layer nodes.

// eslint-disable-next-line @typescript-eslint/no-namespace,@typescript-eslint/no-unused-vars
declare namespace editor {
	export function can_get(node_id: string, property: Property): boolean;

	export function get(
		node_id: string,
		property: Extract<Property, 'id'>,
	): NodeId;

	export function get(
		node_id: string,
		property: Extract<Property, 'type'>,
	): string;

	export function get(
		node_id: string,
		property: Extract<Property, 'children' | 'components' | 'shapes'>,
	): NodeId[];

	export type Property = 'id' | 'type' | 'children' | 'components' | 'shapes';

	export type Location =
		| 'Edit'
		| 'View'
		| 'Project'
		| 'Debug'
		| 'Assets'
		| 'Bundle'
		| 'Scene'
		| 'Outline';

	export type SelectionType = 'resource' | 'outline' | 'scene';

	export type Cardinality = 'one' | 'many';

	export type CommandOptions = {
		selection: NodeId | NodeId[];
	};

	export type NodeId = string & { __brand: 'NodeId' };

	export type EditorCommand = {
		label: string;
		locations: Location[];
		query?: {
			selection: { type: SelectionType; cardinality: Cardinality };
		};
		active?: (opts: CommandOptions) => true;
		run: (opts: CommandOptions) => void;
	};
}
